  public static class FakeCloud { public static Dictionary<string,(string value,string lk)> Store=new Dictionary<string,(string,string)>(); public static bool FailNextLoad, RaceNextSave; public static Action OnLoad; public static int Saves, Loads;
    public static void ExternalWrite(string key,string value){ Store[key]=(value, Guid.NewGuid().ToString("N").Substring(0,8)); } }
  public class PlayerDataService {
    public Task<Dictionary<string,string>> SaveAsync(IDictionary<string,object> d){ FakeCloud.Saves++; var r=new Dictionary<string,string>(); foreach(var kv in d){ FakeCloud.ExternalWrite(kv.Key,(string)kv.Value); r[kv.Key]=FakeCloud.Store[kv.Key].lk; } return Task.FromResult(r); }
    public Task<Dictionary<string,string>> SaveAsync(IDictionary<string,SaveItem> d){ FakeCloud.Saves++; if(FakeCloud.RaceNextSave){ FakeCloud.RaceNextSave=false; FakeCloud.ExternalWrite("save", System.Text.Json.JsonSerializer.Serialize(new CMP5402.Week7.SaveData{ saveCounter=900, saveId="racer", level=50 }, new System.Text.Json.JsonSerializerOptions{IncludeFields=true})); } var r=new Dictionary<string,string>(); foreach(var kv in d){ if(FakeCloud.Store.TryGetValue(kv.Key,out var cur) && kv.Value.WriteLock!=null && cur.lk!=kv.Value.WriteLock) throw new CloudSaveConflictException(); FakeCloud.ExternalWrite(kv.Key,(string)kv.Value.Value); r[kv.Key]=FakeCloud.Store[kv.Key].lk; } return Task.FromResult(r); }
    public Task<Dictionary<string,Item>> LoadAsync(HashSet<string> keys){ FakeCloud.Loads++; FakeCloud.OnLoad?.Invoke(); FakeCloud.OnLoad=null; if(FakeCloud.FailNextLoad){ FakeCloud.FailNextLoad=false; throw new Exception("network down"); } var r=new Dictionary<string,Item>(); foreach(var k in keys) if(FakeCloud.Store.TryGetValue(k,out var v)) r[k]=new Item{ Value=new ItemValue{ S=v.value }, WriteLock=v.lk }; return Task.FromResult(r); }
    public Task DeleteAsync(string key){ FakeCloud.Store.Remove(key); return Task.CompletedTask; } }
